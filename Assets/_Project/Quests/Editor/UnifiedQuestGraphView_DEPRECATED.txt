// UnifiedQuestGraphView v5 — single GraphView-based unified editor for NPCs + Dialogs + Quests.
// Uses QuestGraphModel as the read/write adapter. All connections are object-ref based,
// persisted in the three SOs (NpcDefinition, DialogTree, QuestDefinition).
//
// Architecture: ARCHITECTURE_PLAN.md
// Replaces: v4 UnifiedQuestGraphView (DEPRECATED — see UnifiedQuestGraphView_DEPRECATED.cs)

#if UNITY_EDITOR
using System.Collections.Generic;
using System.Linq;
using UnityEditor;
using UnityEditor.Experimental.GraphView;
using UnityEngine;
using UnityEngine.UIElements;

namespace ProjectC.Quests.Editor
{
    public class UnifiedQuestGraphView : GraphView
    {
        public QuestGraphModel Model { get; } = new();

        private readonly Dictionary<object, BaseGraphNode> _nodeMap = new();
        private readonly Dictionary<EdgeInfo, Edge> _edgeMap = new();
        private bool _editMode;

        private const float NODE_W = 250f;
        private const float H_GAP = 60f;
        private const float V_GAP = 30f;

        public bool EditMode
        {
            get => _editMode;
            set { _editMode = value; MarkDirtyRepaint(); }
        }

        public UnifiedQuestGraphView()
        {
            SetupZoom(0.2f, 2.5f);
            this.AddManipulator(new ContentDragger());
            this.AddManipulator(new SelectionDragger());
            this.AddManipulator(new RectangleSelector());
            this.AddManipulator(new ContentZoomer());
            var grid = new GridBackground();
            Insert(0, grid);
            grid.StretchToParentSize();
            graphViewChanged = OnGraphViewChanged;
        }

        // ── Public API ──

        public void AddNpc(NpcDefinition npc) { Model.AddNpc(npc); Rebuild(); }
        public void AddQuest(QuestDefinition q) { Model.AddQuest(q); Rebuild(); }
        public void AddDialogTree(DialogTree t) { Model.AddDialogTree(t); Rebuild(); }

        public void ClearAll()
        {
            Model.Clear();
            ClearGraphElements();
        }

        public void Rebuild()
        {
            ClearGraphElements();
            Model.BuildGraph();
            BuildVisualGraph();
        }

        public void SaveAll()
        {
            AssetDatabase.SaveAssets();
            Debug.Log($"[UnifiedGraph] Saved all assets. Nodes: {Model.TotalNodeCount}, Edges: {Model.Edges.Count}");
        }

        // ── Build ──

        private void ClearGraphElements()
        {
            foreach (var e in edges.ToList()) RemoveElement(e);
            foreach (var n in nodes.ToList()) RemoveElement(n);
            _nodeMap.Clear();
            _edgeMap.Clear();
        }

        private void BuildVisualGraph()
        {
            // ── Create nodes ──

            foreach (var ni in Model.NpcNodes)
            {
                var node = new NpcGraphNode(ni);
                AddElement(node);
                _nodeMap[ni] = node;
            }

            foreach (var di in Model.DialogNodes)
            {
                var node = new DialogGraphNode(di);
                AddElement(node);
                _nodeMap[di] = node;
            }

            foreach (var qi in Model.QuestNodes)
            {
                var node = new QuestRootGraphNode(qi);
                AddElement(node);
                _nodeMap[qi] = node;
            }

            foreach (var si in Model.StageNodes)
            {
                var node = new StageGraphNode(si);
                AddElement(node);
                _nodeMap[si] = node;
            }

            foreach (var oi in Model.ObjectiveNodes)
            {
                var node = new ObjectiveGraphNode(oi);
                AddElement(node);
                _nodeMap[oi] = node;
            }

            foreach (var ri in Model.RewardNodes)
            {
                var node = new RewardGraphNode(ri);
                AddElement(node);
                _nodeMap[ri] = node;
            }

            // ── Create edges ──
            foreach (var edgeInfo in Model.Edges)
            {
                var edge = CreateVisualEdge(edgeInfo);
                if (edge != null)
                {
                    AddElement(edge);
                    _edgeMap[edgeInfo] = edge;
                }
            }

            // ── Layout ──
            ApplyLayout();
            schedule.Execute(() => { MarkDirtyRepaint(); FrameAll(); }).StartingIn(50);
        }

        private Edge CreateVisualEdge(EdgeInfo info)
        {
            if (!_nodeMap.TryGetValue(info.fromNode, out var fromNode)) return null;
            if (!_nodeMap.TryGetValue(info.toNode, out var toNode)) return null;

            Port fromPort = FindPort(fromNode, Direction.Output, info.fromPort, info.extraIndex);
            Port toPort = FindPort(toNode, Direction.Input, info.toPort, -1);

            if (fromPort == null || toPort == null) return null;

            var edge = fromPort.ConnectTo(toPort);
            edge.viewDataKey = "model-edge";

            // Color coding
            Color c = GraphNodeColors.PortGray;
            switch (info.fromPort)
            {
                case PortSemantic.NpcDefaultDialog: c = GraphNodeColors.PortPurple; break;
                case PortSemantic.NpcOffersQuest: c = GraphNodeColors.PortOrange; break;
                case PortSemantic.DialogEdgeAction: c = GraphNodeColors.PortGreen; break;
                case PortSemantic.ObjectiveTarget: c = GraphNodeColors.PortBlue; break;
                case PortSemantic.StageNext: c = new Color(0.35f, 0.70f, 0.35f); break;
            }
            edge.edgeControl.inputColor = c;
            edge.edgeControl.outputColor = c;
            return edge;
        }

        private static Port FindPort(Node node, Direction dir, PortSemantic semantic, int extraIndex)
        {
            var container = dir == Direction.Output ? node.outputContainer : node.inputContainer;
            foreach (var child in container.Children())
            {
                if (child is Port p && p.direction == dir)
                {
                    var ps = BaseGraphNode.GetSemantic(p);
                    var pd = BaseGraphNode.GetPortData(p);
                    if (ps == semantic)
                    {
                        // If extraIndex matters, match it
                        if (extraIndex >= 0 && pd is int idx && idx != extraIndex) continue;
                        return p;
                    }
                }
            }
            return null;
        }

        // ── Auto-layout ──

        private void ApplyLayout()
        {
            float x = 0f;
            float y = 0f;
            const float NPC_H = 160f;
            const float DLG_H = 270f;
            const float Q_H = 200f;
            const float STAGE_H = 260f;
            const float OBJ_H = 200f;
            const float REWARD_H = 200f;

            // Column 0: NPCs
            foreach (var ni in Model.NpcNodes)
            {
                if (_nodeMap.TryGetValue(ni, out var node))
                    node.SetPosition(new Rect(x, y, NODE_W, NPC_H));
                y += NPC_H + V_GAP;
            }
            x += NODE_W + H_GAP;
            y = 0f;

            // Column 1: Dialog nodes (grouped by tree)
            var treeGroups = Model.DialogNodes.GroupBy(d => d.tree);
            foreach (var group in treeGroups)
            {
                float gy = y;
                foreach (var di in group)
                {
                    if (_nodeMap.TryGetValue(di, out var node))
                        node.SetPosition(new Rect(x, gy, NODE_W, DLG_H));
                    gy += DLG_H + V_GAP;
                }
                y = gy;
            }
            x += NODE_W + H_GAP;
            y = 0f;

            // Column 2+: Quests (each quest occupies its own sub-column)
            foreach (var qi in Model.QuestNodes)
            {
                float qx = x;
                float qy = y;

                if (_nodeMap.TryGetValue(qi, out var qNode))
                    qNode.SetPosition(new Rect(qx, qy, NODE_W, Q_H));
                qy += Q_H + V_GAP;

                // Stages
                var stages = Model.StageNodes.Where(s => s.quest == qi.quest).OrderBy(s => s.stageIndex);
                foreach (var si in stages)
                {
                    if (_nodeMap.TryGetValue(si, out var sNode))
                        sNode.SetPosition(new Rect(qx, qy, NODE_W, STAGE_H));
                    qy += STAGE_H + V_GAP;

                    // Objectives for this stage
                    var objs = Model.ObjectiveNodes.Where(o => o.quest == qi.quest && o.stageIndex == si.stageIndex);
                    foreach (var oi in objs)
                    {
                        if (_nodeMap.TryGetValue(oi, out var oNode))
                            oNode.SetPosition(new Rect(qx + NODE_W + 30, qy - STAGE_H - V_GAP, NODE_W * 0.8f, OBJ_H));
                    }
                }

                // Reward
                var reward = Model.RewardNodes.FirstOrDefault(r => r.quest == qi.quest);
                if (reward != null && _nodeMap.TryGetValue(reward, out var rNode))
                    rNode.SetPosition(new Rect(qx, qy, NODE_W, REWARD_H));

                x += NODE_W + H_GAP + (NODE_W * 0.8f) + H_GAP;
            }
        }

        // ── GraphView callbacks ──

        private GraphViewChange OnGraphViewChanged(GraphViewChange change)
        {
            // Edge creation
            if (change.edgesToCreate != null)
            {
                foreach (var edge in change.edgesToCreate)
                {
                    if (edge != null) HandleEdgeCreated(edge);
                }
            }

            // Edge/node deletion
            if (change.elementsToRemove != null)
            {
                foreach (var elem in change.elementsToRemove)
                {
                    if (elem is Edge edge) HandleEdgeDeleted(edge);
                }
            }

            return change;
        }

        private void HandleEdgeCreated(Edge edge)
        {
            if (edge.output?.node is not BaseGraphNode fromNode ||
                edge.input?.node is not BaseGraphNode toNode)
            {
                // Remove incompatible edges
                if (edge.output != null && edge.input != null)
                    RemoveElement(edge);
                return;
            }

            var fromSem = BaseGraphNode.GetSemantic(edge.output);
            var toSem = BaseGraphNode.GetSemantic(edge.input);
            int edgeIdx = BaseGraphNode.GetPortData(edge.output) is int ei ? ei : -1;

            // Find the model-level node infos
            object fromInfo = GetNodeInfo(fromNode);
            object toInfo = GetNodeInfo(toNode);

            if (fromInfo == null || toInfo == null)
            {
                RemoveElement(edge);
                return;
            }

            // Attempt write-through
            bool ok = Model.SetConnection(fromInfo, fromSem, toInfo, toSem, edgeIdx);
            if (!ok)
            {
                // Connection not supported
                RemoveElement(edge);
                Debug.LogWarning($"[UnifiedGraph] Cannot connect {fromSem} → {toSem}");
                return;
            }

            // Style the edge
            edge.viewDataKey = "user-edge";
            Color c = fromSem switch
            {
                PortSemantic.DialogEdgeAction => GraphNodeColors.PortOrange,
                PortSemantic.ObjectiveTarget => GraphNodeColors.PortBlue,
                PortSemantic.NpcOffersQuest => GraphNodeColors.PortOrange,
                _ => GraphNodeColors.PortGreen,
            };
            edge.edgeControl.inputColor = c;
            edge.edgeControl.outputColor = c;

            // Rebuild dialog node ports if we modified its edges
            if (fromSem == PortSemantic.DialogEdgeAction && fromNode is DialogGraphNode dgn)
            {
                schedule.Execute(() => RebuildSingleDialogNode(dgn)).StartingIn(20);
            }

            Debug.Log($"[UnifiedGraph] Connected {fromSem} → {toSem}");
        }

        private void HandleEdgeDeleted(Edge edge)
        {
            if (edge.output?.node is not BaseGraphNode fromNode ||
                edge.input?.node is not BaseGraphNode toNode) return;

            var fromSem = BaseGraphNode.GetSemantic(edge.output);
            var toSem = BaseGraphNode.GetSemantic(edge.input);
            int edgeIdx = BaseGraphNode.GetPortData(edge.output) is int ei ? ei : -1;

            object fromInfo = GetNodeInfo(fromNode);
            object toInfo = GetNodeInfo(toNode);

            if (fromInfo != null && toInfo != null)
            {
                Model.RemoveConnection(fromInfo, fromSem, toInfo, toSem, edgeIdx);

                if (fromSem == PortSemantic.DialogEdgeAction && fromNode is DialogGraphNode dgn)
                    schedule.Execute(() => RebuildSingleDialogNode(dgn)).StartingIn(20);
            }
        }

        private static object GetNodeInfo(BaseGraphNode node)
        {
            return node switch
            {
                NpcGraphNode n => n.Info,
                DialogGraphNode n => n.Info,
                QuestRootGraphNode n => n.Info,
                StageGraphNode n => n.Info,
                ObjectiveGraphNode n => n.Info,
                RewardGraphNode n => n.Info,
                _ => null
            };
        }

        private void RebuildSingleDialogNode(DialogGraphNode oldNode)
        {
            // Remove old node
            var connectedEdges = edges.ToList().Where(e =>
                (e.output?.node == oldNode) || (e.input?.node == oldNode)).ToList();
            foreach (var e in connectedEdges) RemoveElement(e);
            var pos = oldNode.GetPosition();
            RemoveElement(oldNode);
            _nodeMap.Remove(oldNode.Info);

            // Create new node with updated ports
            var newNode = new DialogGraphNode(oldNode.Info);
            newNode.SetPosition(pos);
            AddElement(newNode);
            _nodeMap[oldNode.Info] = newNode;

            // Rebuild edges for this dialog tree
            Model.BuildGraph();
            foreach (var edgeInfo in Model.Edges)
            {
                if (_nodeMap.TryGetValue(edgeInfo.fromNode, out var fn) &&
                    _nodeMap.TryGetValue(edgeInfo.toNode, out var tn))
                {
                    if (fn == newNode || tn == newNode)
                    {
                        var visEdge = CreateVisualEdge(edgeInfo);
                        if (visEdge != null) AddElement(visEdge);
                    }
                }
            }
            MarkDirtyRepaint();
        }

        // ── Port compatibility ──

        public override List<Port> GetCompatiblePorts(Port startPort, NodeAdapter nodeAdapter)
        {
            var all = new List<Port>();
            foreach (var p in ports)
            {
                if (p == startPort) continue;
                if (p.node == startPort.node) continue;
                if (p.direction == startPort.direction) continue;
                all.Add(p);
            }
            return all;
        }
    }
}
#endif
